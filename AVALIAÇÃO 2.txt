GET = /api/obras/museu/{nome}/		Buscar obras por museu
GET = /api/obras/ano/{ano}/		Buscar obras pelo ano de criação
GET = /api/obras/buscar/?nome={nome}	Pesquisar obras pelo nome

BANCO DE DADOS

CREATE DATABASE db_obras_de_arte;

USE db_obras_de_arte;

TABELA

CREATE TABLE obras_arte (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(150) NOT NULL,
    artista VARCHAR(150) NOT NULL,
    ano_criacao INT,
    movimento_artistico VARCHAR(100),
    tecnica VARCHAR(100),
    museu VARCHAR(150),
    pais_origem VARCHAR(100),
    descricao TEXT,
    valor_estimado DECIMAL(15,2)
);
